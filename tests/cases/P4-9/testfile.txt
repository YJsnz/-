int add(int x, int y){
    return(x + y);
}
void print_if(int n){
    if (n > 0) {
        printf("pos=", n);
    } else {
        printf("nonpos=", n);
    }
}

    sum = 0;
    for (i = 0; i < 3; i = i + 1) {
        sum = sum + add(i, 10);
    }
    printf("sum=", sum);
    print_if(sum);
    i = 0;
    while (i < 2) {
        i = i + 1;
